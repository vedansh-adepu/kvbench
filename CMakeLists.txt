function(kvbench_options target)
  target_compile_options(${target} PRIVATE
    $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wsign-conversion;-Wshadow>
    $<$<CXX_COMPILER_ID:MSVC>:/W4>)
  if(KVBENCH_WARNINGS_AS_ERRORS)
    target_compile_options(${target} PRIVATE
      $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Werror>
      $<$<CXX_COMPILER_ID:MSVC>:/WX>)
  endif()
  if(KVBENCH_SANITIZERS)
    if(MSVC)
      message(FATAL_ERROR "Combined ASan/UBSan preset requires GCC or Clang")
    endif()
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
  endif()
  if(KVBENCH_COVERAGE)
    if(MSVC)
      message(FATAL_ERROR "Coverage preset requires GCC or Clang")
    endif()
    target_compile_options(${target} PRIVATE --coverage -O0 -g)
    target_link_options(${target} PRIVATE --coverage)
  endif()
endfunction()

if(KVBENCH_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Fuzz targets require Clang/libFuzzer")
  endif()
  if(BUILD_TESTING)
    message(FATAL_ERROR "Configure fuzz builds with BUILD_TESTING=OFF")
  endif()
  target_compile_options(kvbench_core PRIVATE -fsanitize=fuzzer-no-link,undefined -fno-sanitize-recover=all)
  foreach(kind IN ITEMS config vllm)
    add_executable(fuzz_${kind} fuzz/${kind}_fuzzer.cpp)
    target_link_libraries(fuzz_${kind} PRIVATE kvbench_core)
    kvbench_options(fuzz_${kind})
    target_compile_options(fuzz_${kind} PRIVATE -fsanitize=fuzzer-no-link,undefined -fno-sanitize-recover=all)
    if(KVBENCH_FUZZ_RUNTIME)
      if(NOT EXISTS "${KVBENCH_FUZZ_RUNTIME}")
        message(FATAL_ERROR "KVBENCH_FUZZ_RUNTIME archive not found")
      endif()
      target_link_libraries(fuzz_${kind} PRIVATE "${KVBENCH_FUZZ_RUNTIME}")
      target_link_options(fuzz_${kind} PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
    else()
      target_link_options(fuzz_${kind} PRIVATE -fsanitize=fuzzer,undefined -fno-sanitize-recover=all)
    endif()
  endforeach()
endif()
